klines.forEach((kline, i) => { const timestamp = kline[0]; // K 線開盤時間 (ms) const date = new Date(timestamp); const dayOfWeek = date.getUTCDay(); // 0 = 週日, 1 = 週一 ... 6 = 週六 const hour = date.getUTCHours(); const minute = date.getUTCMinutes(); let isLineTime = false; let lineColor = 'rgba(0, 240, 255, 0.35)'; // 預設明顯亮青色 if (timeframe === '1d') { // 【日週期】：每 1 週區間（每週一 00:00 UTC）畫一條，強制使用黃色 if (dayOfWeek === 1 && hour === 0 && minute === 0) { isLineTime = true; lineColor = 'rgba(255, 220, 0, 0.75)'; // 明顯黃色 } } else if (timeframe === '4h') { // 【4小時】：每 1 日區間（每日 00:00 UTC）畫一條 if (hour === 0 && minute === 0) { isLineTime = true; lineColor = 'rgba(0, 240, 255, 0.4)'; // 明顯亮青色 } } else if (timeframe === '1h' || timeframe === '15m') { // 【1小時 / 15分】：每 1 日區間（每日 00:00 UTC）畫一條 if (hour === 0 && minute === 0) { isLineTime = true; // 該日若為週一，則該條日虛線改用黃色 if (dayOfWeek === 1) { lineColor = 'rgba(255, 220, 0, 0.7)'; // 週一用黃色 } else { lineColor = 'rgba(0, 240, 255, 0.35);'; // 一般日用明顯亮青色 } } } if (isLineTime) { const x = getXCoordinate(i); ctx.save(); ctx.strokeStyle = lineColor; ctx.lineWidth = 1; ctx.setLineDash([3, 3]); // 虛線樣式 ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, chartHeight); ctx.stroke(); ctx.restore(); } });
